    <script>
        const searchInput = document.getElementById('searchInput');
        const engineSelect = document.getElementById('engineSelect');
        const quickCards = document.querySelectorAll('.quick-card');

        // Функция поиска с учётом выбранной системы
        function performSearch(query) {
            let q = query.trim();
            if (!q) return;

            // Если это ссылка — открываем напрямую
            if (
                q.startsWith('http://') ||
                q.startsWith('https://') ||
                (q.includes('.') && !q.includes(' ') && (q.includes('.com') || q.includes('.ru') || q.includes('.org') || q.includes('.net') || q.includes('/')))
            ) {
                let url = q;
                if (!url.startsWith('http')) url = 'https://' + url;
                window.location.href = url;
                return;
            }

            const engine = engineSelect.value;
            let searchUrl = '';

            switch (engine) {
                case 'google':
                    searchUrl = `https://www.google.com/search?q=${encodeURIComponent(q)}`;
                    break;
                case 'yandex':
                    searchUrl = `https://ya.ru/search/?text=${encodeURIComponent(q)}`;
                    break;
                case 'bing':
                    searchUrl = `https://www.bing.com/search?q=${encodeURIComponent(q)}`;
                    break;
                case 'duckduckgo':
                    searchUrl = `https://duckduckgo.com/?q=${encodeURIComponent(q)}`;
                    break;
                default:
                    searchUrl = `https://www.google.com/search?q=${encodeURIComponent(q)}`;
            }

            window.location.href = searchUrl;
        }

        // Enter — поиск
        searchInput.addEventListener('keypress', (e) => {
            if (e.key === 'Enter') {
                performSearch(searchInput.value);
            }
        });

        // Клик по иконке лупы — тоже поиск
        const searchIcon = document.querySelector('.search-icon');
        if (searchIcon) {
            searchIcon.addEventListener('click', () => {
                performSearch(searchInput.value);
            });
        }

        // При смене поисковика обновляем placeholder
        engineSelect.addEventListener('change', () => {
            const engineNames = {
                google: 'Поиск в Google или введите URL',
                yandex: 'Поиск в Яндексе или введите URL',
                bing: 'Поиск в Bing или введите URL',
                duckduckgo: 'Поиск в DuckDuckGo или введите URL',
            };
            searchInput.placeholder = engineNames[engineSelect.value] || 'Введите поисковый запрос или URL';
        });

        // Обработка кликов по быстрым ссылкам (карточкам)
        quickCards.forEach(card => {
            card.addEventListener('click', (e) => {
                const url = card.getAttribute('data-url');
                if (url) {
                    window.location.href = url;
                }
            });
        });

        // Языковые ссылки (демо-режим)
        document.querySelectorAll('.lang-link').forEach(link => {
            link.addEventListener('click', (e) => {
                const lang = e.target.getAttribute('data-lang');
                alert(`🌐 Язык интерфейса изменён на ${lang === 'ru' ? 'Русский' : 'English'} (демо-режим).`);
            });
        });

        // Автофокус на поиск
        searchInput.focus();

        // Дополнительный эффект для карточек — звуковое сопровождение не требуется, всё в стиле
        console.log("CoreFlow — цветовая гамма в точности как в QStyleSheet: #37271C, #4F3625, #362112, #E37225 и другие оттенки.");
    </script>
